function filteredGpsPoint(raw,accuracy,speed,timestamp){
 if(!gps)return {...raw};
 const now=Number.isFinite(timestamp)&&timestamp>0?timestamp:Date.now();
 const elapsed=Math.max(0.25,Math.min(5,(now-(gpsSampleTime||now-1000))/1000));
 const distance=meters(gps,raw);
 const accuracyAllowance=Math.max(12,(Number.isFinite(accuracy)?accuracy:25)*1.8);
 const plausibleSpeed=Number.isFinite(speed)&&speed>=0?speed:0;
 const maxPlausibleJump=Math.max(35,plausibleSpeed*elapsed*2.8+accuracyAllowance);
 // Descarta saltos incompatíveis com o intervalo e a velocidade reportados pelo GPS.
 if(distance>maxPlausibleJump&&accuracy<=GPS_MAX_ACCURACY_METERS)return {...gps};
 // Precisão ruim não deve deslocar bruscamente o marcador.
 if(Number.isFinite(accuracy)&&accuracy>GPS_MAX_ACCURACY_METERS)return {...gps};
 const accuracyWeight=Number.isFinite(accuracy)?clamp((30-accuracy)/25,0.12,0.58):0.18;
 const speedWeight=plausibleSpeed>8?0.18:plausibleSpeed>2?0.08:0;
 const alpha=clamp(accuracyWeight+speedWeight,0.16,0.68);
 return {lat:gps.lat+(raw.lat-gps.lat)*alpha,lon:gps.lon+(raw.lon-gps.lon)*alpha};
}

function bearing(a,b){const p1=rad(a.lat),p2=rad(b.lat),dl=rad(b.lon-a.lon),y=Math.sin(dl)*Math.cos(p2),x=Math.cos(p1)*Math.sin(p2)-Math.sin(p1)*Math.cos(p2)*Math.cos(dl);return(Math.atan2(y,x)*180/Math.PI+360)%360}

function pointSegmentDistanceMeters(point,a,b){
 const lat0=rad(point.lat),scaleX=111320*Math.cos(lat0),scaleY=110540;
 const px=(point.lon)*scaleX,py=point.lat*scaleY;
 const ax=a[0]*scaleX,ay=a[1]*scaleY,bx=b[0]*scaleX,by=b[1]*scaleY;
 const dx=bx-ax,dy=by-ay;
 const t=Math.max(0,Math.min(1,((px-ax)*dx+(py-ay)*dy)/(dx*dx+dy*dy||1)));
 return Math.hypot(px-(ax+t*dx),py-(ay+t*dy));
}
function distanceFromRoute(point,coords){

 }
 return min;
}

function routeProgressAtPosition(point,coordinates,minAlong=0,maxAlong=Infinity){
 if(!point||!Array.isArray(coordinates)||coordinates.length<2)return null;
 const lengths=[0];let total=0;
 for(let i=1;i<coordinates.length;i++){
  total+=meters({lat:coordinates[i-1][1],lon:coordinates[i-1][0]},{lat:coordinates[i][1],lon:coordinates[i][0]});
  lengths.push(total);
 }
 if(total<=0)return null;
 let bestDistance=Infinity,bestAlong=null;
 const latScale=Math.cos(point.lat*Math.PI/180);
 for(let i=1;i<coordinates.length;i++){
  const segmentStart=lengths[i-1],segmentEnd=lengths[i];
  if(segmentEnd<minAlong||segmentStart>maxAlong)continue;
  const a=coordinates[i-1],b=coordinates[i];
  const ax=(a[0]-point.lon)*111320*latScale,ay=(a[1]-point.lat)*110540;
  const bx=(b[0]-point.lon)*111320*latScale,by=(b[1]-point.lat)*110540;
  const dx=bx-ax,dy=by-ay,den=dx*dx+dy*dy;
  const t=den?Math.max(0,Math.min(1,-(ax*dx+ay*dy)/den)):0;
  const px=ax+t*dx,py=ay+t*dy,d=Math.hypot(px,py);
  const along=segmentStart+t*(segmentEnd-segmentStart);
  if(along<minAlong||along>maxAlong)continue;
  if(d<bestDistance){bestDistance=d;bestAlong=along;}
 }
 if(bestAlong===null)return null;
 return {progress:Math.max(0,Math.min(1,bestAlong/total)),along:bestAlong,total,distance:bestDistance};
}

function routeMainRoads(route,maxRoads=5){
 const steps=(route?.legs||[]).flatMap(leg=>leg.steps||[]);
 const names=[];
 for(const step of steps){
  const name=String(step?.name||step?.ref||step?.pronunciation||"").trim();
  if(!name||/^(unnamed road|unknown road|sem nome|via sem nome)$/i.test(name))continue;
  if(names[names.length-1]!==name&&!names.includes(name))names.push(name);
 }
 return {names:names.slice(0,maxRoads),total:names.length};
}
